App.tsx
import React, { useState } from "react";
import ChatWindow from "./components/ChatWindow";
import ChatInput from "./components/ChatInput";
import "./styles/ChatApp.css";

export interface Message {
  sender: "user" | "bot";
  content: string;
}

const App: React.FC = () => {
  const [messages, setMessages] = useState<Message[]>([]);

  const handleSendMessage = async (message: string) => {
    // Add user message to chat
    setMessages((prev) => {
      const updatedMessages = [...prev, { sender: "user", content: message }];
      console.log("Updated messages with user message:", updatedMessages);
      return updatedMessages;
    });

    // Call the backend API to get the bot's response
    try {
      const response = await fetch("http://localhost:8000/readque/", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ query: message }),
      });

      if (!response.ok) {
        throw new Error("Error fetching response from server");
      }

      const data = await response.json();
      console.log("botmsg:", data);
      const botMessage = data;

      // Add bot message to chat
      setMessages((prev) => {
        const updatedMessages = [...prev, { sender: "bot", content: botMessage }];
        console.log("Updated messages with bot response:", updatedMessages);
        return updatedMessages;
      });
    } catch (error) {
      console.error("Error:", error);
    }
  };

  console.log("messages:", messages);

  return (
    <div className="chat-app">
      <ChatWindow messages={messages} />
      <ChatInput onSend={handleSendMessage} />
    </div>
  );
};

export default App;

ChatWindow.tsx:
import React from "react";
import { Message } from "../App";

interface ChatWindowProps {
  messages: Message[];
}

const ChatWindow: React.FC<ChatWindowProps> = ({ messages }) => {
  return (
    
    <div className="chat-window">
      {messages.map((msg, index) => (
        <div key={index} className={`chat-message ${msg.sender}`}>
          {msg.content}
          
        </div>
      ))}
    </div>
  );
};

export default ChatWindow;

ChatInput.tsx:
import React, { useState } from "react";

interface ChatInputProps {
  onSend: (message: string) => void;
}

const ChatInput: React.FC<ChatInputProps> = ({ onSend }) => {
  const [input, setInput] = useState("");

  const handleSend = () => {
    if (input.trim()) {
      onSend(input);
      setInput("");
    }
  };

  return (
    <div className="chat-input">
      <input
        type="text"
        value={input}
        onChange={(e) => setInput(e.target.value)}
        placeholder="Type a message..."
      />
      <button onClick={handleSend}>Send</button>
    </div>
  );
};

export default ChatInput;
